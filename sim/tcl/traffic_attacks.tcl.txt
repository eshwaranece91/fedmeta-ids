proc start_ddos {node_id rate} {
    global ns
    set udp [new Agent/UDP]
    $ns attach-agent $node_($node_id) $udp
    set cbr [new Application/Traffic/CBR]
    $cbr set rate_ ${rate}Mb
    $cbr attach-agent $udp
    $ns at 10.0 "$cbr start"
}
